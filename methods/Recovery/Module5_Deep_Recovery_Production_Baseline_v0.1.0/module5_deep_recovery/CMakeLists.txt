cmake_minimum_required(VERSION 3.20)
project(deep_recovery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DEEP_BUILD_TESTS "Build tests" ON)
add_library(deepcore src/source.cpp src/detector.cpp src/scanner.cpp src/score.cpp src/state.cpp src/json.cpp)
target_include_directories(deepcore PUBLIC include)
if(MSVC)
 target_compile_options(deepcore PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(deepcore PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
endif()
add_executable(deepscan src/main.cpp)
target_link_libraries(deepscan PRIVATE deepcore)
if(DEEP_BUILD_TESTS)
 enable_testing()
 add_executable(deep_tests tests/test_main.cpp)
 target_link_libraries(deep_tests PRIVATE deepcore)
 add_test(NAME deep_tests COMMAND deep_tests)
endif()
install(TARGETS deepscan RUNTIME DESTINATION bin)
