cmake_minimum_required(VERSION 3.20)
project(storage_raid_recovery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RAID_BUILD_TESTS "Build tests" ON)

add_library(raidcore
 src/source.cpp src/layout.cpp src/mdraid.cpp src/raid5.cpp src/raid6.cpp
 src/raid10.cpp src/score.cpp src/json.cpp)
target_include_directories(raidcore PUBLIC include)
if(MSVC)
 target_compile_options(raidcore PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(raidcore PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
endif()
add_executable(raidscan src/main.cpp)
target_link_libraries(raidscan PRIVATE raidcore)
if(RAID_BUILD_TESTS)
 enable_testing()
 add_executable(raid_tests tests/test_main.cpp)
 target_link_libraries(raid_tests PRIVATE raidcore)
 add_test(NAME raid_tests COMMAND raid_tests)
 add_executable(raid_recovery_tests tests/test_raid_recovery.cpp)
 target_link_libraries(raid_recovery_tests PRIVATE raidcore)
 add_test(NAME raid_recovery_tests COMMAND raid_recovery_tests)
endif()
install(TARGETS raidscan RUNTIME DESTINATION bin)
