cmake_minimum_required(VERSION 3.20)
project(targeted_recovery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TARGETED_BUILD_TESTS "Build tests" ON)
add_library(targeted_core
 src/source.cpp src/signature.cpp src/scanner.cpp src/validator.cpp src/recovery.cpp src/json.cpp)
target_include_directories(targeted_core PUBLIC include)
if(MSVC)
 target_compile_options(targeted_core PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(targeted_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(targetedscan src/main.cpp)
target_link_libraries(targetedscan PRIVATE targeted_core)
if(TARGETED_BUILD_TESTS)
 enable_testing()
 add_executable(targeted_tests tests/test_main.cpp)
 target_link_libraries(targeted_tests PRIVATE targeted_core)
 add_test(NAME targeted_tests COMMAND targeted_tests)
endif()
install(TARGETS targetedscan RUNTIME DESTINATION bin)
