cmake_minimum_required(VERSION 3.20)
project(forensic_recovery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FORENSIC_BUILD_TESTS "Build tests" ON)
add_library(forensiccore src/sha256.cpp src/evidence.cpp src/case.cpp src/acquire.cpp src/report.cpp)
target_include_directories(forensiccore PUBLIC include)
if(MSVC)
 target_compile_options(forensiccore PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(forensiccore PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
endif()
add_executable(forensicctl src/main.cpp)
target_link_libraries(forensicctl PRIVATE forensiccore)
if(FORENSIC_BUILD_TESTS)
 enable_testing()
 add_executable(forensic_tests tests/test_main.cpp)
 target_link_libraries(forensic_tests PRIVATE forensiccore)
 add_test(NAME forensic_tests COMMAND forensic_tests)
endif()
install(TARGETS forensicctl RUNTIME DESTINATION bin)
