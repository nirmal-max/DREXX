cmake_minimum_required(VERSION 3.20)
project(damaged_media_recovery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MEDIA_BUILD_TESTS "Build tests" ON)
add_library(mediacore src/source.cpp src/imager.cpp src/map.cpp src/strategy.cpp src/json.cpp)
target_include_directories(mediacore PUBLIC include)
if(MSVC)
 target_compile_options(mediacore PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(mediacore PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
endif()
add_executable(mediaimager src/main.cpp)
target_link_libraries(mediaimager PRIVATE mediacore)
if(MEDIA_BUILD_TESTS)
 enable_testing()
 add_executable(media_tests tests/test_main.cpp)
 target_link_libraries(media_tests PRIVATE mediacore)
 add_test(NAME media_tests COMMAND media_tests)
endif()
install(TARGETS mediaimager RUNTIME DESTINATION bin)
