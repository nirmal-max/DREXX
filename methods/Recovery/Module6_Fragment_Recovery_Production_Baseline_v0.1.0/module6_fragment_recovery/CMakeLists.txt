cmake_minimum_required(VERSION 3.20)
project(fragment_recovery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FRAGMENT_BUILD_TESTS "Build tests" ON)
add_library(fragmentcore src/source.cpp src/signatures.cpp src/chains.cpp src/score.cpp src/reconstruct.cpp src/json.cpp)
target_include_directories(fragmentcore PUBLIC include)
if(MSVC)
 target_compile_options(fragmentcore PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(fragmentcore PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
endif()
add_executable(fragmentscan src/main.cpp)
target_link_libraries(fragmentscan PRIVATE fragmentcore)
if(FRAGMENT_BUILD_TESTS)
 enable_testing()
 add_executable(fragment_tests tests/test_main.cpp)
 target_link_libraries(fragment_tests PRIVATE fragmentcore)
 add_test(NAME fragment_tests COMMAND fragment_tests)
endif()
install(TARGETS fragmentscan RUNTIME DESTINATION bin)
